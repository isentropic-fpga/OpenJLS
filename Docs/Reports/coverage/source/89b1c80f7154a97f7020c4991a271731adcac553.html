   328:       Write(ManagerRec, A_HEIGHT, x"00000004");

   335:       SetBurstMode(StreamTxRec, STREAM_BURST_BYTE_MODE);

   338:

   350:       -- Phase 3: APPLY mid-image aborts the encode; the next image is clean

   352:       -- Feed half an image and let the core consume it, then APPLY. The aborted 
   353:       -- encode has emitted only whole OUT_WIDTH beats (never TLAST), so the 
   354:       -- receiver holds that residue until the post-abort image's TLAST closes 
   355:       -- the burst. The fresh golden stream must sit byte-exact at the tail. 
   356:       PushBurst(StreamTxRec.BurstFifo, H3_PIXELS(0 to H3_PIXELS'length / 2 - 1)); 
   357:       SendBurstAsync(StreamTxRec, H3_PIXELS'length / 2);

   360:       WaitForClock(ManagerRec, 4); 
   361:

   370:                & to_string(vResidue) & " bytes)"); 
   371:       for i in 0 to numBytes - 1 loop 
   372:         Pop(StreamRxRec.BurstFifo, rxByte);

   375:                         H3_EXPECTED(i - vResidue), 
   376:                         "post-abort byte " & to_string(i - vResidue)); 
   377:         end if; 
   378:       end loop; 
   379:
