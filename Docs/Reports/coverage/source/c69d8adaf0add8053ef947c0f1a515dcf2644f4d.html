    10:   -- Module Name: byte_stuffer - Behavioral 
    11:   -- Description: 
    12:   -- 
    13:   --   Per T.87: every 0xFF byte in the encoded bitstream must be followed by a 
    14:   --   stuffed '0' bit so decoders can distinguish payload from markers (an FF 
    15:   --   followed by a byte with MSB='1' denotes a marker). 
    16:   -- 
    17:   --   Three internal stages: 
    18:   -- 
    19:   --     Stage 1 — bit packer: 
    20:   --       Accumulates input bits MSB-first into a 2*FIFO_BITS accumulator. 
    21:   --       Drains a fixed FIFO_BITS-wide word into the FIFO whenever enough 
    22:   --       bits are present. On flush the sub-byte residue is padded to a 
    23:   --       byte boundary; the final FIFO write is always FIFO_BITS wide 
    24:   --       (zero-padded if needed) and carries the final word's real valid-bit 
    25:   --       count (the byte-boundary pad excluded) plus last_flag=1. Stage 3 
    26:   --       counts only those real bits, so the pad is never emitted; the 
    27:   --       genuine residue is padded post-stuffing at the terminal beat. 
    28:   -- 
    29:   --     Stage 2 — BRAM-backed sync FIFO 
    30:   -- 
    31:   --     Stage 3 — FF stuffer + output emit: 
    32:   --       Refills a holding register using the nine fixed alignments allowed 
    33:   --       by the refill contract (0..8 old bits). Each cycle forms up to four 
    34:   --       output bytes by decoding all eight legal stuffing layouts in 
    35:   --       parallel. Each layout uses fixed bit slices and constant valid-bit 
    36:   --       thresholds. One-hot selection chooses the output, the last FF state, 
    37:   --       and a fixed-shift remainder without a serial lane-resolution chain. 
    38:   -- 
    39:   --       The end-of-image terminal beat (sub-byte residue, a pending stuff 
    40:   --       bit with no follow-up data, or a byte-aligned clean end) is split 
    41:   --       into its own cycle via sLastPending: the final byte (or 0-byte beat) 
    42:   --       is assembled, latched, and emitted on the following beat. Adds at 
    43:   --       most 1 cycle of latency per image boundary and keeps the pad-byte 
    44:   --       assembly off the critical path. 
    45:   -- 
    46:   --   Flush protocol (iFlush, single-cycle pulse from upstream on the cycle 
    47:   --   the bit_packer presents the image's last word): 
    48:   --     - Stage 1 pads its sub-byte residue and tags the final FIFO write 
    49:   --       with last_flag=1. 
    50:   --     - Stage 3 latches the last_flag when it pops that word. Once the 
    51:   --       holding register drains, it inserts a final stuff '0' if the last 
    52:   --       payload byte was 0xFF, zero-pads the output accumulator to a byte 
    53:   --       boundary, emits the remaining whole bytes, and pulses oFlushDone 
    54:   --       on the final output beat (oFlushDone is sampled together with 
    55:   --       oWordValid='1', matching jls_framer's iEOI contract). 
    56:   -- 
    57:   -- Generics: 
    58:   --   IN_WIDTH            : bit_packer worst-case word width (= LIMIT). 
    59:   --   OUT_BYTES_PER_CYCLE : output bytes/cycle; must be 4 (fixed layout table). 
    60:   --   BURST_DEPTH         : depth of the BRAM-backed FIFO (in wide words).

   357:               end loop;

   359:             end if;

   654:    
   655:           ---------------------------------------------------------------------- 
   656:           -- (3) Decode all layouts in parallel. Each predicate depends only on 
   657:           --     the incoming FF state and fixed-window comparisons, never on 
   658:           --     a byte or consumption count selected by another lane. 
   659:           ---------------------------------------------------------------------- 
   660:           vPath(0) := not vPrevFF and not ff0 and not ff1b and not ff2b; 
   661:           vPath(1) := not vPrevFF and not ff0 and not ff1b and     ff2b; 
   662:           vPath(2) := not vPrevFF and not ff0 and     ff1b; 
   663:           vPath(3) := not vPrevFF and     ff0 and not ff2a; 
   664:           vPath(4) := not vPrevFF and     ff0 and     ff2a; 
   665:           vPath(5) :=     vPrevFF and not ff1a and not ff2a; 
   666:           vPath(6) :=     vPrevFF and not ff1a and     ff2a; 
   667:           vPath(7) :=     vPrevFF and     ff1a; 
   668:    
   669:           vEmitData      := (others => '0'); 
   670:           vEmitBytesBits := (others => '0'); 
   671:           vEmitLastFF    := '0'; 
   672:           vTakeShift     := (others => '0'); 
   673:    
   674:           for p in STUFF_LAYOUT'range loop 
   675:    
   676:             vCandidateWord := (others => '0'); 
   677:             vPathMask      := (others => vPath(p)); 
   678:    
   679:             for lane in 0 to OUT_BYTES_PER_CYCLE - 1 loop 
   680:    
   681:               if (STUFF_LAYOUT(p)(lane) = '1') then 
   682:                 vCandidateByte := '0' & vStuffBuffer(HOLD_BITS - 1 - LANE_START(p, lane) 
   683:                                                      downto HOLD_BITS - LANE_END(p, lane)); 
   684:               else 
   685:                 vCandidateByte := vStuffBuffer(HOLD_BITS - 1 - LANE_START(p, lane) 
   686:                                               downto HOLD_BITS - LANE_END(p, lane)); 
   687:               end if; 
   688:               vCandidateWord(OUT_WIDTH - 1 - lane * 8 downto OUT_WIDTH - (lane + 1) * 8) := vCandidateByte; 
   689:    
   690:               -- Select the last complete lane using constant thresholds. The 
   691:               -- end thresholds increase strictly, so exactly one lane wins. 
   692:               vLaneSelected := vPath(p) and iReady and 
   693:                                bool2bit(vStuffBufferBits >= LANE_END(p, lane)); 
   694:               if (lane < OUT_BYTES_PER_CYCLE - 1) then 
   695:                 vLaneSelected := vLaneSelected and 
   696:                                  bool2bit(vStuffBufferBits < LANE_END(p, lane + 1)); 
   697:               end if; 
   698:               vBytesMask     := (others => vLaneSelected); 
   699:               vEmitBytesBits := vEmitBytesBits or (to_unsigned(lane + 1, vEmitBytesBits'length) and vBytesMask); 
   700:               vEmitLastFF    := vEmitLastFF or (vLaneSelected and bool2bit(vCandidateByte = x"FF")); 
   701:    
   702:               for k in CONSUME_BITS'range loop 
   703:    
   704:                 if (CONSUME_BITS(k) = LANE_END(p, lane)) then 
   705:                   vTakeShift(k) := vTakeShift(k) or vLaneSelected; 
   706:                 end if; 
   707:    
   708:               end loop; 
   709:    
   710:             end loop; 
   711:    
   712:             vEmitData := vEmitData or (vCandidateWord and vPathMask); 
   713:    
   714:           end loop; 
   715:    
   716:           ---------------------------------------------------------------------- 
   717:           -- (4) Select fixed shifts and constant count decrements in parallel. 
   718:           --     No general barrel shifter follows a late selected integer count. 
   719:           --     Include the no-emission case so stalls preserve all input bits. 
   720:           ---------------------------------------------------------------------- 
   721:           vEmitBytes := to_integer(vEmitBytesBits); 
   722:           -- No complete first byte: this decision does not depend on the 
   723:           -- layout decoder or the selected output count. 
   724:           vTakeShift(0) := not iReady or bool2bit(vStuffBufferBits < 7) or 
   725:                            (not vPrevFF and bool2bit(vStuffBufferBits = 7)); 
   726:           vNextBuffer := (others => '0'); 
   727:           vNextCount  := (others => '0'); 
   728:    
   729:           for k in CONSUME_BITS'range loop 
   730:    
   731:             vBufferMask := (others => vTakeShift(k)); 
   732:             vCountMask  := (others => vTakeShift(k)); 
   733:             vNextBuffer := vNextBuffer or 
   734:                            (std_logic_vector(shift_left(unsigned(vStuffBuffer), CONSUME_BITS(k))) and vBufferMask); 
   735:             -- Unsigned arithmetic deliberately wraps on unselected candidates 
   736:             -- with too few bits. Their mask is zero; only a legal count wins. 
   737:             vNextCount := vNextCount or 
   738:                           ((to_unsigned(vStuffBufferBits, vNextCount'length) - CONSUME_BITS(k)) and vCountMask); 
   739:    
   740:           end loop; 
   741:    
   742:           vStuffBuffer     := vNextBuffer; 
   743:           vStuffBufferBits := to_integer(vNextCount); 
   744:           if (vEmitBytes > 0) then 
   745:             vPrevFF := vEmitLastFF; 
   746:           end if; 
   747:    
   748:           ---------------------------------------------------------------------- 
   749:           -- (5) Output register and flush-done / drain entry. 
   750:           ---------------------------------------------------------------------- 
   751:           if (vEmitBytes > 0) then 
   752:             sOutWordReg       <= vEmitData; 
   753:             sOutBytesValidReg <= to_unsigned(vEmitBytes, sOutBytesValidReg'length); 
   754:             sOutValidReg      <= '1'; 
   755:           else 
   756:             sOutValidReg <= '0'; 
   757:           end if; 
   758:    
   759:           -- Once the last word is consumed and only a sub-byte residue remains 
   760:           -- (bits < 8, including the bits=0 clean end), hand off to the 
   761:           -- sLastPending branch which assembles the final beat off this path. 
   762:           if (iReady = '1' 
   763:               and vStuffBufferLast = '1' 
   764:               and vStuffBufferBits < 8) then 
   765:             sLastPending <= '1'; 
   766:           end if; 
   767:    
   768:           sStuffBuffer     <= vStuffBuffer; 
   769:           sStuffBufferBits <= to_unsigned(vStuffBufferBits, sStuffBufferBits'length); 
   770:           sStuffBufferLast <= vStuffBufferLast; 
   771:           sPrevFF          <= vPrevFF; 
   772:         end if; 
   773:       end if; 
   774:    
   775:     end process stage3_proc; 
   776:    
   777:   end architecture behavioral;
