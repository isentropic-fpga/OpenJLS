    10:   -- T.87 Code segment A.23 (standard form), transcribed verbatim: 
    11:   --   if (Errval < 0) Nn += 1; 
    12:   --   A += (EMErrval + 1 - RItype) >> 1;          -- floor shift 
    13:   --   if (N == RESET) { A>>=1; N>>=1; Nn>>=1; } 
    14:   --   N += 1; 
    15:   -- where EMErrval = 2*abs(Errval) - RItype - map (A.22). The RTL drops map via the 
    16:   -- Mert Fig.9 equivalence (A += abs(Errval) - RItype). The reference deliberately 
    17:   -- keeps the *standard* map-dependent form and drives map as a free random input 
    18:   -- not fed to the DUT, so a passing check also proves the equivalence holds for 
    19:   -- both map values. 
    20:   -- 
    21:   -- Every result is representable for T.87-coherent inputs: 
    22:   --   A_new = A + |Errval| - RItype >= 0: RItype=0 => A+|Errval|>=0; RItype=1 => 
    23:   --     |Errval|>=1 (a run-interruption sample has Ix/=Ra, see tb_a22) => A_new>=A>=0. 
    24:   --   Nn_new <= RESET-1: the invariant N-Nn>=1 holds from init (N=1,Nn=0) and is 
    25:   --     preserved by the update (N+1, Nn+<=1) and the rescale (floor halving), so 
    26:   --     input Nn<=N-1 and Nn_new = Nn+1 <= N <= RESET-1 (N=RESET takes the halving).
