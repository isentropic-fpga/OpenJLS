    11:   -- Description: 
    12:   -- 
    13:   --    Concatenates this cycle's raw + Golomb fields into a single 
    14:   --    variable-length word. Output is (oWord, oValidLen): MSB-aligned 
    15:   --    bitstream in oWord, oValidLen counts how many top bits are 
    16:   --    meaningful. Downstream byte_stuffer accumulates these 
    17:   --    variable-length words and produces byte-aligned output. 
    18:   -- 
    19:   --    Word layout (MSB → LSB): 
    20:   --      [raw bits (rawLen)][unary zeros (unaryZeros)]['1'][suffix (suffixLen)] 
    21:   -- 
    22:   --    When only one of raw/Golomb fires, the corresponding portion is 
    23:   --    omitted. Both fire simultaneously in RI (run-interruption) mode. 
    24:   -- 
    25:   --    No internal buffer. Concat is combinational on the inputs; 
    26:   --    outputs are registered to break the comb chain into byte_stuffer.
