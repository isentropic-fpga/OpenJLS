    10:   -- Architectural block (no T.87 C segment). Contract verified against an 
    11:   -- independent behavioral model: 
    12:   --   * RdLatency = 1. 
    13:   --   * First read of an address since reset/EOI returns CTX_INIT (the packed 
    14:   --     A_INIT|B_INIT|C_INIT|N_INIT word), regardless of prior writes; the read 
    15:   --     clears the init flag for that address. 
    16:   --   * Same-cycle write to the read address forwards iWrData (WBR rebuilt over the 
    17:   --     RBW BRAM) -- but only once the address is past its init read (init wins). 
    18:   --   * Otherwise the read returns the last written value (RBW: a same-cycle write 
    19:   --     to a *different* path does not affect this read). 
    20:   --   * Reset / EndOfImage re-arms every address to CTX_INIT (BRAM contents kept). 
    21:   --   * A read issued ON the iEndOfImage cycle (the pipeline does this: EOI rides 
    22:   --     with the last pixel, whose context read is in flight that same cycle) 
    23:   --     resolves against the ENDING image's init flags; the re-arm applies after. 
    24:   --     Regression for the boundary-image bug where this read returned raw BRAM 
    25:   --     for a first-use context (k=0 -> spurious escape on the last pixel). 
    26:   --   * Accesses CAN be in flight on the first reset cycle (the pipeline's 
    27:   --     valid registers clear synchronously, so the pre-reset valid still 
    28:   --     drives the enables that cycle). They must be harmless: the re-arm wins 
    29:   --     over the read's flag-clear, a write lands in BRAM but stays masked by 
    30:   --     the re-armed init flag, and the read result is a don't-care (every 
    31:   --     downstream consumer is reset on the cycle it would be used). 
    32:   -- Every address is written before any second read so the modelled and real 
    33:   -- BRAM contents always agree. 
    34:   -- Coverage closes the three read outcomes plus fresh/seen reads on EOI cycles.

   353:       -- Directed: accesses in flight on the FIRST reset cycle (found by the 
   354:       -- top-level mid-stream reset stress; the module previously assumed them 
   355:       -- away). The re-arm must win over the in-flight read's flag-clear, and 
   356:       -- the in-flight write must stay masked until its post-reset init read. 
   357:       -------------------------------------------------------------------------- 
   358:       d := std_logic_vector(to_unsigned(7777, TOTAL_WIDTH)); 
   359:       step('1', 40, '1', 40, d, "addr40 init read + write-back"); 
   360:       step('1', 40, '0', 0, ZERO, "addr40 = written value"); 
   361:       d := std_logic_vector(to_unsigned(9999, TOTAL_WIDTH)); 
   362:       pulse_rst_access(40, 55, d); 
   363:       step('1', 40, '0', 0, ZERO, "addr40 after reset-with-read-in-flight = CTX_INIT"); 
   364:       step('1', 55, '0', 0, ZERO, "addr55 written during reset = CTX_INIT (init masks)"); 
   365:       step('1', 55, '0', 0, ZERO, "addr55 second read = value written during reset"); 
   366:    
   367:       -------------------------------------------------------------------------- 
   368:       -- Directed regression (boundary-image bug): a read in flight ON the EOI 
   369:       -- cycle. The preceding read is a seen address, so the buggy version's 
   370:       -- stale flag served raw BRAM for the fresh address below. 
   371:       -------------------------------------------------------------------------- 
   372:       d := std_logic_vector(to_unsigned(3333, TOTAL_WIDTH)); 
   373:       step('1', 21, '1', 21, d, "addr21 init read + write-back"); 
   374:       step('1', 21, '0', 0, ZERO, "addr21 = written value"); 
   375:       -- Fresh address read on the EOI cycle: must be CTX_INIT, not raw BRAM. 
   376:       step('1', 33, '0', 0, ZERO, "fresh addr33 on EOI cycle = CTX_INIT", '1'); 
   377:       -- Re-armed by that EOI: addr 21 is fresh again despite its BRAM value. 
   378:       step('1', 21, '0', 0, ZERO, "addr21 after EOI = CTX_INIT"); 
   379:       -- Seen address read on the EOI cycle: must be the BRAM value (the buggy 
   380:       -- version's stale flag could equally serve CTX_INIT here). 
   381:       step('1', 21, '0', 0, ZERO, "seen addr21 on EOI cycle = BRAM value", '1'); 
   382:       step('1', 21, '0', 0, ZERO, "addr21 after second EOI = CTX_INIT"); 
   383:    
   384:       -------------------------------------------------------------------------- 
   385:       -- Constrained-random: random addresses, always read-modify-write so the 
   386:       -- modelled and real BRAM stay in sync. 
   387:       -------------------------------------------------------------------------- 
   388:       -- Initialize every physical address with unique data, including 365/366. 
   389:       pulse_eoi; 
   390:       for addr in 0 to RAM_DEPTH - 1 loop 
   391:         d := std_logic_vector(to_unsigned(addr + 1, TOTAL_WIDTH)); 
   392:         step('1', addr, '1', addr, d, "address sweep first use"); 
   393:       end loop; 
   394:       for addr in 0 to RAM_DEPTH - 1 loop 
   395:         step('1', addr, '0', 0, ZERO, "address sweep retained value"); 
   396:       end loop; 
   397:       for addr in 0 to RAM_DEPTH - 1 loop 
   398:         d := rv.RandSlv(TOTAL_WIDTH); 
   399:         step('1', addr, '1', (addr + 1) mod RAM_DEPTH, d, "independent read/write"); 
   400:         step('0', (addr + 2) mod RAM_DEPTH, '1', addr, not d, "write-only cycle"); 
   401:         step('0', addr, '0', addr, ZERO, "idle cycle"); 
   402:       end loop; 
   403:       step('1', 366, '1', 366, d, "forward on EOI", '1'); 
   404:       step('1', 366, '0', 0, ZERO, "EOI rearms forwarded address"); 
   405:    
   406:       for i in 1 to 4000 loop 
   407:    
   408:         a := rv.RandInt(0, RAM_DEPTH - 1); 
   409:         d := rv.RandSlv(TOTAL_WIDTH); 
   410:    
   411:         -- ~1-in-40 reads ride an EOI cycle (the last-pixel pattern); otherwise 
   412:         -- plain read-modify-write. 
   413:         if (rv.DistValInt(((1, 1), (0, 40))) = 1) then 
   414:           step('1', a, '1', a, d, "rand rmw+eoi a=" & integer'image(a), '1'); 
   415:         else 
   416:           step('1', a, '1', a, d, "rand rmw a=" & integer'image(a)); 
   417:         end if; 
   418:    
   419:         -- Occasional idle-cycle EOI or mid-stream iRst (with or without an 
   420:         -- access in flight on the first reset cycle) as well. 
   421:         if (rv.DistValInt(((1, 1), (0, 60))) = 1) then 
   422:           pulse_eoi; 
   423:         elsif (rv.DistValInt(((1, 1), (0, 80))) = 1) then 
   424:           pulse_rst; 
   425:         elsif (rv.DistValInt(((1, 1), (0, 100))) = 1) then 
   426:           pulse_rst_access(rv.RandInt(0, RAM_DEPTH - 1), 
   427:                            rv.RandInt(0, RAM_DEPTH - 1), rv.RandSlv(TOTAL_WIDTH)); 
   428:         end if; 
   429:    
   430:         exit when IsCovered(cov) and IsCovered(covEoiRd) and i > 200; 
   431:    
   432:       end loop; 
   433:

   435:       WriteBin(covEoiRd); 
   436:       AffirmIf(GetAlertLogID("CoverageClosure"), IsCovered(cov), "read-outcome coverage closed"); 
   437:       AffirmIf(GetAlertLogID("CoverageClosure"), IsCovered(covEoiRd), "read-on-EOI-cycle coverage closed"); 
   438:    
   439:       end_of_test("tb_context_ram_osvvm");
