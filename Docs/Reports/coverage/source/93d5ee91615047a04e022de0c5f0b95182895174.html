   467:       procedure send_beats ( 
   468:         beats : natural 
   469:       ) is

   477:    
   478:           -- ~30% all-ones words for stuff runs 
   479:           if (rv.DistValInt(((1, 3), (0, 7))) = 1) then

   491:    
   492:             end loop;

   494:           end if;

   520:       -- No wait between the images, within the top's limits: 3 in flight, and >= 4 words 
   521:       -- per image so stage 1's flush clears. Merging loses a flush-done, so the wait is bounded.
