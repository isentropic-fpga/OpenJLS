   406:       -- the terminal type for coverage, and reset for the next image. 
   407:    
   408:       procedure finish_image is 
   409:    
   410:         variable ft : integer; 
   411:    
   412:       begin 
   413:    
   414:         if (not anyBits) then 
   415:           ft := FT_EMPTY; 
   416:         elsif (bitsInByte = 0) then 
   417:           ft := FT_CLEAN; 
   418:         elsif (bitsInByte = 1 and lastEmitFF and not realSince) then 
   419:           ft := FT_DANGLING; 
   420:         else 
   421:           ft := FT_RESIDUE; 
   422:         end if; 
   423:    
   424:         if (bitsInByte > 0) then 
   425:           Push(SB_ID, curByte);                -- unfilled low bits are already '0' (post-stuffing pad) 
   426:         end if; 
   427:    
   428:         ICover(covFlush, ft); 
   429:    
   430:         curByte    := (others => '0'); 
   431:         bitsInByte := 0; 
   432:         anyBits    := false; 
   433:         lastEmitFF := false; 
   434:         realSince  := false; 
   435:         stuffSeen  := false; 
   436:    
   437:       end procedure finish_image; 
   438:    
   439:       -- Present one beat and advance only when iStall='0' on a rising edge 
   440:       -- (== one DUT latch/consume; bit_packer holds its output across a stall). 
   441:    
   442:       procedure send_beat ( 
   443:         w     : std_logic_vector; 
   444:         len   : natural; 
   445:         valid : std_logic; 
   446:         flush : std_logic 
   447:       ) is 
   448:       begin 
   449:    
   450:         iWord         <= w; 
   451:         iWordValidLen <= to_unsigned(len, VLEN_W); 
   452:         iWordValid    <= valid; 
   453:         iFlush        <= flush; 
   454:    
   455:         loop 
   456:    
   457:           wait until rising_edge(clk); 
   458:           exit when iStall = '0'; 
   459:    
   460:         end loop; 
   461:    
   462:         -- consumed on this edge; update reference model in DUT order (append then flush) 
   463:         if (valid = '1') then 
   464:           push_word(w, len); 
   465:         end if; 
   466:    
   467:         if (flush = '1') then 
   468:           finish_image; 
   469:         end if; 
   470:    
   471:         iWordValid <= '0'; 
   472:         iFlush     <= '0'; 
   473:    
   474:       end procedure send_beat; 
   475:    
   476:       -- One image: random beats; flush rides the LAST valid word (the DUT 
   477:       -- contract -- iFlush pulses on the cycle the final word is presented, never 
   478:       -- on an empty beat). Then wait for full drain before the next image. 
   479:    
   480:       procedure send_image ( 
   481:         beats : natural 
   482:       ) is 
   483:    
   484:         variable w   : std_logic_vector(IN_WIDTH - 1 downto 0); 
   485:         variable len : natural; 
   486:    
   487:       begin 
   488:    
   489:         for n in 1 to beats loop 
   490:    
   491:           -- ~30% all-ones words to manufacture 0xFF output bytes and stuff runs. 
   492:           if (rv.DistValInt(((1, 3), (0, 7))) = 1) then 
   493:             w := (others => '1'); 
   494:           else 
   495:             w := (others => '0'); 
   496:    
   497:             for k in 0 to IN_WIDTH / 8 - 1 loop 
   498:    
   499:               if (rv.DistValInt(((1, 1), (0, 7))) = 1) then 
   500:                 w(IN_WIDTH - 1 - k * 8 downto IN_WIDTH - (k + 1) * 8) := x"FF"; 
   501:               else 
   502:                 w(IN_WIDTH - 1 - k * 8 downto IN_WIDTH - (k + 1) * 8) := rv.RandSlv(0, 255, 8); 
   503:               end if; 
   504:    
   505:             end loop; 
   506:    
   507:           end if; 
   508:    
   509:           len := rv.RandInt(1, IN_WIDTH); 
   510:    
   511:           if (n = beats) then 
   512:             send_beat(w, len, '1', '1');   -- flush rides the final word 
   513:           else 
   514:             send_beat(w, len, '1', '0'); 
   515:           end if; 
   516:    
   517:         end loop; 
   518:    
   519:         sImagesSent <= sImagesSent + 1; 
   520:         -- serialize images: wait for full drain before the next one 
   521:         wait until sFlushDone = sImagesSent; 
   522:    
   523:       end procedure send_image; 
   524:    
   525:       -- Directed beat helper: a single word of `len` valid bits. 
   526:    
   527:       procedure directed ( 
   528:         w   : std_logic_vector(IN_WIDTH - 1 downto 0); 
   529:         len : natural 
   530:       ) is 
   531:       begin 
   532:    
   533:         send_beat(w, len, '1', '1'); 
   534:         sImagesSent <= sImagesSent + 1; 
   535:         wait until sFlushDone = sImagesSent; 
   536:    
   537:       end procedure directed; 
   538:    
   539:       -- Drive n full beats (flush on the last if flush='1'), then iRst. The 
   540:       -- aborted output is never scored (sIgnore); after reset the module must be 
   541:       -- idle and a fresh image byte-exact. The reference state is clean here 
   542:       -- (last image was flushed) and is re-initialised to mirror the reset. 
   543:       procedure reset_mid ( 
   544:         n     : natural; 
   545:         flush : std_logic 
   546:       ) is 
   547:       begin 
   548:    
   549:         sIgnore <= true; 
   550:    
   551:         for k in 1 to n loop 
   552:    
   553:           iWord         <= rv.RandSlv(IN_WIDTH); 
   554:           iWordValidLen <= to_unsigned(IN_WIDTH, VLEN_W); 
   555:           iWordValid    <= '1'; 
   556:           if (k = n) then 
   557:             iFlush <= flush; 
   558:           else 
   559:             iFlush <= '0'; 
   560:           end if; 
   561:    
   562:           loop 
   563:    
   564:             wait until rising_edge(clk); 
   565:             exit when iStall = '0'; 
   566:    
   567:           end loop; 
   568:    
   569:         end loop; 
   570:    
   571:         iWordValid <= '0'; 
   572:         iFlush     <= '0'; 
   573:         if (n = 0) then 
   574:           ICover(covRst, 0); 
   575:         else 
   576:           ICover(covRst, 1 + std_to_int(flush)); 
   577:         end if; 
   578:         apply_reset(clk, rst, 6, '1'); 
   579:         wait for 1 ns; 
   580:         AffirmIf(GetAlertLogID("ResetRecovery"), oWordValid = '0', "mid-op reset: byte_stuffer output cleared"); 
   581:         AffirmIf(GetAlertLogID("ResetRecovery"), oFlushDone = '0', "mid-op reset: no flush-done after reset"); 
   582:    
   583:         curByte    := (others => '0'); 
   584:         bitsInByte := 0; 
   585:         anyBits    := false; 
   586:         lastEmitFF := false; 
   587:         realSince  := false; 
   588:         stuffSeen  := false; 
   589:    
   590:         sIgnore <= false; 
   591:         wait until rising_edge(clk); 
   592:    
   593:         send_image(rv.RandInt(1, MAX_BEATS)); 
   594:    
   595:       end procedure reset_mid; 
   596:    
   597:       variable ones : std_logic_vector(IN_WIDTH - 1 downto 0); 
   598:       variable wtmp : std_logic_vector(IN_WIDTH - 1 downto 0); 
   599:    
   600:     begin 
   601:    
   602:       -- Stim-driven inputs (no signal defaults; assign before reset). 
   603:       iWord         <= ZERO_W; 
   604:       iWordValid    <= '0'; 
   605:       iWordValidLen <= (others => '0'); 
   606:       iFlush        <= '0'; 
   607:    
   608:       -- Reference FF-stuffer state. 
   609:       curByte    := (others => '0'); 
   610:       bitsInByte := 0; 
   611:       anyBits    := false; 
   612:       lastEmitFF := false; 
   613:       realSince  := false; 
   614:       stuffSeen  := false; 
   615:    
   616:       SetAlertLogName("tb_byte_stuffer_osvvm"); 
   617:       SetLogEnable(PASSED, FALSE); 
   618:       rv.InitSeed(rv'instance_name); 
   619:    
   620:       covEmit := NewID("G4.Stream emitBytes"); 
   621:       SetFieldName(covEmit, "emitBytes"); 
   622:       for count in 0 to OUT_BYTES loop 
   623:         AddBins(covEmit, "emitted bytes=" & to_string(count), GenBin(count)); 
   624:       end loop; 
   625:       -- FT_EMPTY is unreachable (see directed-corner note) and excluded. 
   626:       covFlush := NewID("G4.Stream flushType"); 
   627:       SetFieldName(covFlush, "flushType"); 
   628:       AddBins(covFlush, "byte aligned", GenBin(1)); 
   629:       AddBins(covFlush, "padded residue", GenBin(2)); 
   630:       AddBins(covFlush, "dangling FF", GenBin(3)); 
   631:       -- Require seeing data output BOTH with and without almost-full backpressure. 
   632:       -- (The 0-byte terminal beat only fires after the FIFO has drained, so 
   633:       -- 0-byte x almostFull is unreachable and deliberately not a bin.) 
   634:       covCross := NewID("G4.Stream data x almostFull"); 
   635:       SetFieldName(covCross, "data", "almostFull"); 
   636:       for almost_full in 0 to 1 loop 
   637:         AddCross(covCross, "data beat / almostFull=" & to_string(almost_full), 
   638:                  GenBin(1), GenBin(almost_full)); 
   639:       end loop; 
   640:       -- Stuff event per emitted byte: at least one image with no stuff, one with 
   641:       -- its first stuff, and one with a subsequent (run) stuff. Makes the FF->'0' 
   642:       -- stuffing -- the module's whole purpose -- visible in its own coverage. 
   643:       covStuff := NewID("G4.Branch stuffEvent"); 
   644:       SetFieldName(covStuff, "stuffEvent"); 
   645:       AddBins(covStuff, "noStuff",    GenBin(SE_NONE, SE_NONE)); 
   646:       AddBins(covStuff, "firstStuff", GenBin(SE_FIRST, SE_FIRST)); 
   647:       AddBins(covStuff, "runStuff",   GenBin(SE_RUN, SE_RUN)); 
   648:    
   649:       covRst := NewID("G5.States resetPhase"); 
   650:       AddBins(covRst, "idle", GenBin(0)); 
   651:       AddBins(covRst, "mid-image", GenBin(1)); 
   652:       AddBins(covRst, "flush draining", GenBin(2)); 
   653:    
   654:       apply_reset(clk, rst, 6, '1'); 
   655:       ones := (others => '1'); 
   656:    
   657:       -------------------------------------------------------------------------- 
   658:       -- Directed corners (regression locks). 
   659:       -------------------------------------------------------------------------- 
   660:       -- NOTE: a truly-empty flush (zero payload bits) is intentionally NOT tested 
   661:       -- -- it is unreachable. Every valid word into byte_stuffer carries >=1 bit 
   662:       -- (regular/RI: the Golomb unary stop bit in the bit_packer; token_raw: 
   663:       -- A15_A16 guarantees raw suffix len >=1), and flush rides the last valid 
   664:       -- word, so the accumulator is never empty at flush. 
   665:    
   666:       -- Single non-FF byte, byte-aligned clean end. 
   667:       wtmp                                   := (others => '0'); 
   668:       wtmp(IN_WIDTH - 1 downto IN_WIDTH - 8) := x"5A"; 
   669:       directed(wtmp, 8); 
   670:    
   671:       -- Single 0xFF byte -> trailing stuffed 0x00 (dangling-FF terminal). 
   672:       wtmp                                   := (others => '0'); 
   673:       wtmp(IN_WIDTH - 1 downto IN_WIDTH - 8) := x"FF"; 
   674:       directed(wtmp, 8); 
   675:    
   676:       -- Sub-byte residue (5 real bits) padded post-stuffing. 
   677:       wtmp                                   := (others => '0'); 
   678:       wtmp(IN_WIDTH - 1 downto IN_WIDTH - 5) := "10101"; 
   679:       directed(wtmp, 5); 
   680:    
   681:       for len in 1 to IN_WIDTH loop 
   682:         directed(ones, len); 
   683:         directed(ZERO_W, len); 
   684:       end loop; 
   685:       send_beat(ones, IN_WIDTH, '0', '0'); -- inactive garbage must not enter stream 
   686:       directed(ZERO_W, 8); 
   687:    
   688:       -- Long 0xFF run in one beat: forces consecutive stuffs across the 4-slot 
   689:       -- chain (full-width all-ones). 
   690:       directed(ones, IN_WIDTH); 
   691:    
   692:       -- Multi-word flush with residue: > FIFO width of all-ones, flushed on a 
   693:       -- final partial beat (regression for the multi-word valid-bits overflow). 
   694:       send_beat(ones, IN_WIDTH, '1', '0'); 
   695:       send_beat(ones, IN_WIDTH, '1', '0'); 
   696:       wtmp                                    := (others => '0'); 
   697:       wtmp(IN_WIDTH - 1 downto IN_WIDTH - 12) := x"FF" & "0101"; 
   698:       directed(wtmp, 12); 
   699:    
   700:       -- Back-to-back tiny images (context reset between flushes). 
   701:       for n in 1 to 4 loop 
   702:    
   703:         wtmp                                   := (others => '0'); 
   704:         wtmp(IN_WIDTH - 1 downto IN_WIDTH - 8) := std_logic_vector(to_unsigned(n * 32, 8)); 
   705:         directed(wtmp, 8); 
   706:    
   707:       end loop; 
   708:    
   709:       -------------------------------------------------------------------------- 
   710:       -- Constrained-random images. 
   711:       -------------------------------------------------------------------------- 
   712:       for img in 1 to N_RANDOM loop 
   713:    
   714:         send_image(rv.RandInt(1, MAX_BEATS)); 
   715:    
   716:       end loop; 
   717:    
   718:       -------------------------------------------------------------------------- 
   719:       -- Mid-operation iRst: mid-image, idle, and while a flush drains. 
   720:       -------------------------------------------------------------------------- 
   721:       reset_mid(4, '0'); 
   722:       reset_mid(0, '0'); 
   723:       reset_mid(4, '1'); 
   724:    
   725:       -------------------------------------------------------------------------- 
   726:       -- Done: every image already drained (each send_image/directed waits for its 
   727:       -- own oFlushDone), so just let the monitor settle, then report. 
   728:       -------------------------------------------------------------------------- 
   729:       sDriverDone <= true; 
   730:       wait for 10 * CLK_PERIOD; 
   731:    
   732:       AffirmIf(GetAlertLogID("ScoreboardCompletion"), IsEmpty(SB_ID), "scoreboard drained (all expected bytes consumed)"); 
   733:       AffirmIfEqual(GetAlertLogID("ScoreboardCompletion"), GetErrorCount(SB_ID), 0, "scoreboard mismatches"); 
   734:    
   735:       WriteBin(covEmit); 
   736:       WriteBin(covFlush); 
   737:       WriteBin(covCross); 
   738:       WriteBin(covStuff); 
   739:       AffirmIf(GetAlertLogID("CoverageClosure"), IsCovered(covEmit), "emitBytes coverage closed"); 
   740:       AffirmIf(GetAlertLogID("CoverageClosure"), IsCovered(covFlush), "flushType coverage closed"); 
   741:       AffirmIf(GetAlertLogID("CoverageClosure"), IsCovered(covCross), "data x almostFull coverage closed"); 
   742:       AffirmIf(GetAlertLogID("CoverageClosure"), IsCovered(covStuff), "stuff-event coverage closed"); 
   743:    
   744:       end_of_test("tb_byte_stuffer_osvvm"); 
   745:       wait; 
   746:    
   747:       affirm_covered(covRst, "reset phase coverage closed"); 
   748:     end process stim_proc;
